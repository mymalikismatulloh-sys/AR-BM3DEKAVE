  <script>
    const video = document.querySelector('#my-video');
    const button = document.querySelector('#overlay-button');
    const scene = document.querySelector('a-scene');

    // Event listener untuk mendeteksi ketika scene AR siap
    scene.addEventListener("ready", (event) => {
      console.log("Scene AR siap!");
    });

    // Event listener untuk mendeteksi target ditemukan/hilang (opsional untuk logika tambahan)
    const target = document.querySelector('[mindar-image-target]');
    target.addEventListener("targetFound", event => {
        console.log("Target kartu nama ditemukan!");
        // Autoplay video saat target ditemukan (browser mungkin memblokir ini)
        video.play().catch(error => {
            console.log("Autoplay diblokir browser, menampilkan tombol.");
            button.style.display = 'block';
        });
    });

    target.addEventListener("targetLost", event => {
        console.log("Target kartu nama hilang.");
        video.pause(); // Jeda video saat target hilang
    });

    // Logika tombol overlay untuk memulai video jika autoplay diblokir
    button.addEventListener('click', function() {
      video.play();
      button.style.display = 'none'; // Sembunyikan tombol setelah diketuk
    });

  </script>
